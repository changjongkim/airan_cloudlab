% Second TikZ style of the Antiphase design figures (2026-10-11): the look of the ScaleQsim paper, as used by the
% SectorQ design figures (pauliprop/figures/src/tikz_style2.tex).  Used by src/design2_<name>.tex, built by
% build_v2.sh into <name>.png.  The first style (paper/backstop_slot_v14/figures/tikz_style.tex) is kept.
%
% What makes the look:
%   - all text is bold Times; names of panels and components are large, annotations are bold italic
%   - a part of the system is a tinted panel (pan) with its name at the top left; inside it, white component
%     boxes (comp) with a name and a black numbered circle (num)
%   - state is drawn as tables with a blue header row (th) and example values (td), or as rows of colored cells (cl)
%   - stages are joined by fat black block arrows (barrow); thin arrows (arr) inside a component; an action is a
%     yellow rounded box (act); a decision is a mint box (dec)
%   - red (dred) only for the key values of a figure and for what Antiphase stops or drops
% One color per entity: AI blue, NeuralRx green, conventional receiver teal, two-user cell gold,
% TB of slot 2731 violet, 2732 gold, 2733 orange.
% Canvas: 17.6 cm wide.  No text or arrow crosses the border of a box it does not belong to; little empty space.
\definecolor{pan}{HTML}{D9D8C4}     % panel
\definecolor{pan2}{HTML}{EDEDED}    % a lighter panel inside a figure
\definecolor{navy}{HTML}{1F3864}
\definecolor{thblue}{HTML}{0070C0}  % table header
\definecolor{dred}{HTML}{C00000}
\definecolor{dgreen}{HTML}{2E7D32}
\definecolor{actc}{HTML}{FFE699}    % an action
\definecolor{tune}{HTML}{D5F2E3}    % a decision
\definecolor{aic}{HTML}{8FAADC}     % AI
\definecolor{ailarge}{HTML}{2F5597} % a large AI unit
\definecolor{nrxc}{HTML}{A9D18E}    % NeuralRx
\definecolor{convc}{HTML}{C5E0E6}   % conventional receiver
\definecolor{gold}{HTML}{FFC000}    % two-user cell, TB of slot 2732
\definecolor{tba}{HTML}{CDB5E8}     % TB of slot 2731
\definecolor{recc}{HTML}{F4B183}    % TB of slot 2733
\definecolor{badc}{HTML}{FBE0DD}    % stopped, dropped
\definecolor{gpuc}{HTML}{DDDAB8}
