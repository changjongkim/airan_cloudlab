% One uplink TB in one period of the evaluated P180/D155 test timeline, split by
% the NeuralRx outcome. In both cases, the cuPHY owner receives the TB, hands it
% to the NeuralRx worker at once, and holds a pending recovery until the 45 ms
% NeuralRx deadline. (a) The NeuralRx result passes the CRC: the owner commits
% it, and the recovery of the TB is cancelled, which frees its GPU time. (b) The
% result fails the CRC, is late, or is missing: the owner hands the same TB to
% cuPHY recovery on the recovery GPU, which must finish before the 153 ms
% recovery deadline. Until 45 ms the two cases look the same.

\begin{tikzpicture}[x=1cm,y=1cm]
  \def\swT#1{3.3+(#1)*0.0556}

  \draw[guard] ({\swT{153}},1.0) -- ({\swT{153}},5.75);
  \node[font=\itshape\tiny, text=dred, anchor=south] at ({\swT{153}},5.75) {recovery deadline};

  % Case titles and rows.
  \node[font=\bfseries\scriptsize, anchor=west] at (0.0,5.62) {(a) NeuralRx passes the CRC by 45 ms};
  \node[font=\bfseries\scriptsize, anchor=west] at (0.0,3.32) {(b) NeuralRx fails the CRC or is late};
  \foreach \y/\t in {5.0/{cuPHY (TB owner)}, 4.38/{NeuralRx receiver}, 3.76/{Recovery GPU},
                     2.7/{cuPHY (TB owner)}, 2.08/{NeuralRx receiver}, 1.46/{Recovery GPU}} {
    \node[pttl, anchor=west] at (0.1,\y) {\t};
    \draw[black!25, line width=0.4pt] ({\swT{0}},\y) -- ({\swT{185}},\y);
  }

  % holds a pending recovery until the NeuralRx deadline.

    \node[font=\scriptsize, text=navyf] at ({\swT{0}},\o+0.3) {$\blacktriangledown$};

    \node[job, pattern=north east lines, pattern color=gold!90!black, draw=gold!80!black,
          minimum height=0.34cm, minimum width={45*0.0556 cm}, anchor=west] at ({\swT{0}},\o) {};

    \draw[arr] ({\swT{2.5}},\o-0.17) -- ({\swT{2.5}},\n+0.17);
    \node[job, fill=peach, minimum height=0.34cm, minimum width={15*0.0556 cm}, anchor=west]

    \node[font=\itshape\scriptsize, anchor=west, fill=white, inner sep=1pt] at ({\swT{19}},\n) {TensorRT};
    \draw[arr] ({\swT{17}},\n+0.17) -- ({\swT{17}},\o-0.17);
  }

  % (a) The owner commits the NeuralRx result, and the recovery is cancelled.
  \node[font=\scriptsize, text=green!40!black, anchor=west] at ({\swT{18}},4.69) {\cmark\ CRC pass};
  \node[comp, fill=okc, draw=green!45!black, minimum height=0.34cm, inner sep=1pt, anchor=west]
    at ({\swT{47}},5.0) {\cmark\ commit NeuralRx result};
  \node[job, dashed, draw=black!45, fill=aic!35, minimum height=0.34cm,
        minimum width={(153-45)*0.0556 cm}, anchor=west] at ({\swT{45}},3.76) {};
  \node[font=\itshape\scriptsize] at ({\swT{99}},3.76) {recovery cancelled: this GPU time can run AI};
  \node[font=\scriptsize, text=navyf] at ({\swT{180}},5.3) {$\blacktriangledown$};
  \node[font=\itshape\tiny, anchor=east] at ({\swT{180}-0.1},5.31) {next TB};

  % (b) The owner hands the same TB to cuPHY recovery on the recovery GPU.
  \node[font=\scriptsize, text=dred, anchor=west] at ({\swT{18}},2.39) {\xmark\ CRC fail};
  \node[job, dashed, draw=dred, fill=white, minimum height=0.34cm,

  \draw[arr, draw=dred] ({\swT{48}},2.53) -- ({\swT{48}},1.63);
