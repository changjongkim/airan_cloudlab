% Enforcing one AI slot on the GPU. (a) The normal path of one AI slot across
% the coordinator, the marker files, the Qwen worker, and the GPU stream. (b)-(e)
% The injected faults of the C161 fault campaign, drawn as the same kind of
% message sequence: (b) the coordinator holds the run request past the latest
% start (phase-1 latest_start_nonlaunch), (c) the reply comes after the response
% timeout although the GPU finished (post_fence_reply_delay), (d) the worker
% closes the channel after submission and before any completion marker
% (pre_fence_channel_loss), and (e) repeated and old radio results
% (stale_duplicate_nrx, stale_duplicate_recovery). Red marks the fault; the box
% at the bottom of each panel is what the coordinator does.
\resizebox{\linewidth}{!}{%
\begin{tikzpicture}[x=1cm,y=1cm,
    pframe/.style={draw=black!45, line width=0.45pt},
    ptitle/.style={font=\bfseries\scriptsize, anchor=north west},
    life/.style={black!45, dashed, line width=0.5pt},
    mhead/.style={comp, fill=panel, font=\scriptsize, minimum width=1.3cm, minimum height=0.36cm},
    lbl/.style={inote, anchor=south, fill=white, inner sep=1pt},
    fnote/.style={font=\itshape\scriptsize, text=dred},
    good/.style={comp, fill=okc, draw=green!45!black, minimum height=0.36cm},
    kept/.style={comp, fill=debtc, minimum height=0.36cm},
    gnote/.style={font=\itshape\scriptsize, text=green!40!black},
    nnote/.style={font=\itshape\scriptsize, text=black!55},
    lost/.style={font=\bfseries\small, text=dred}]

  % ============================================================ (a) normal path
  \draw[pframe] (0,0) rectangle (13.9,-4.5);
  \node[ptitle] at (0.05,-0.03) {(a) Normal path of one AI slot};
  \foreach \x/\t/\w in {2.3/{Coordinator (host)}/2.3, 5.4/{Marker files}/1.7,

    \node[comp, fill=panel, minimum width=\w cm, minimum height=0.42cm] at (\x,-0.62) {\t};
    \draw[life] (\x,-0.83) -- (\x,-4.3);
  }

  % (1) Commit an AI slot with an absolute latest start time.
  \node[comp, fill=aic, minimum height=0.36cm] at (2.3,-1.1) {commit AI slot $L$, version $g{+}1$};

  % (2) Run request.
  \draw[arr] (2.3,-1.6) -- (8.5,-1.6);
  \node[note, anchor=south, fill=white, inner sep=1pt] at (5.4,-1.58) {run($L$, latest start)};
  \node[num] at (3.4,-1.6) {2};

  % (3) Launch check on the worker; a late request is panel (b).
  \node[comp, minimum width=2.1cm, minimum height=0.34cm] at (8.5,-2.0) {now $\le$ latest start?};
  \node[num] at (9.85,-2.0) {3};

  \node[inote, anchor=south] at (10.75,-2.4) {yes: submit};
  \node[job, fill=aic, minimum width=0.42cm, minimum height=0.8cm] at (11.6,-2.8) {};
  \node[inote, anchor=west, align=left] at (11.88,-2.75) {prefill\\kernels};
  \draw[arr] (8.5,-2.6) -- (6.03,-2.6);
  \node[cell, fill=white, minimum width=1.25cm, anchor=center] at (5.4,-2.6) {L.launch};
  \node[inote, anchor=south] at (7.25,-2.6) {write};

  % (4) GPU completion: CUDA event, completion marker, reply.
  \draw[arr] (11.6,-3.2) -- (8.5,-3.2);
  \node[inote, anchor=south] at (10.05,-3.2) {CUDA event done};
  \node[num] at (12.05,-3.2) {4};
  \draw[arr] (8.5,-3.4) -- (6.03,-3.4);
  \node[cell, fill=okc, minimum width=1.25cm, anchor=center] at (5.4,-3.4) {L.complete};

  \draw[arr] (8.5,-3.75) -- (2.3,-3.75);

  % (5) Free the slot only on a matching completion marker.
  \node[good] at (2.3,-4.15) {free $L$ on marker, version $g{+}2$};
  \node[num] at (0.35,-4.15) {5};

  % ====================================================== fault panels (b)-(e)
  % Panels (b)-(d) use the lifelines of (a) in the same order.
  \foreach \ox/\oy/\t in {0/-4.65/{(b) Run request held past its latest start},
                          7.05/-4.65/{(c) Reply late or lost after the GPU finished},
                          0/-8.35/{(d) AI channel lost before the GPU finished}} {
    \draw[pframe] (\ox,\oy) rectangle ++(6.85,-3.55);

      \draw[life] (\ox+\dx,\oy-0.8) -- (\ox+\dx,\oy-2.85);

  \draw[arr, dashed, draw=dred] (0.95,-5.8) -- node[sloped, above, fnote, inner sep=1pt]

  \node[lbl] at (2.6,-6.98) {reply ``not launched''};
  \node[nnote, fill=white, inner sep=1pt] at (5.9,-7.0) {no kernel};
  \node[good, anchor=west] at (0.1,-7.55) {free $L$ at once; AI goes on};

  % (c) The GPU finished and wrote L.complete, but the reply does not arrive.
  \draw[arr] (8.0,-5.8) -- (11.3,-5.8);
  \node[lbl] at (9.65,-5.78) {run($L$)};
  \draw[arr] (11.3,-6.02) -- (12.95,-6.02);
  \node[inote, anchor=south] at (12.12,-6.02) {submit};

  \draw[arr] (12.95,-6.4) -- (11.3,-6.4);
  \node[inote, anchor=south] at (12.12,-6.4) {done};
  \draw[arr] (11.3,-6.6) -- (10.28,-6.6);
  \node[cell, fill=okc, minimum width=1.25cm, anchor=center] at (9.65,-6.6) {L.complete};

  \node[good, anchor=west] at (7.15,-7.85) {free $L$ from the marker; stop AI};

  % (d) The channel closes after launch and before any completion marker.
  \draw[arr] (0.95,-9.5) -- (4.25,-9.5);

  \draw[arr] (4.25,-9.75) -- (5.9,-9.75);
  \node[inote, anchor=south] at (5.07,-9.75) {submit};
  \node[job, fill=aic, minimum width=0.36cm, minimum height=1.45cm] at (5.9,-10.475) {};

  \draw[arr] (4.25,-9.97) -- (3.23,-9.97);
  \node[cell, fill=white, minimum width=1.25cm, anchor=center] at (2.6,-9.97) {L.launch};
  \draw[dred, line width=0.8pt] (4.25,-10.45) -- (1.47,-10.45);
  \node[lost] at (1.35,-10.45) {\xmark};
  \node[fnote, anchor=south, fill=white, inner sep=1pt] at (3.35,-10.43) {channel closed};
  \draw[arr] (0.95,-10.85) -- (2.6,-10.85);
  \node[lbl] at (1.95,-10.83) {read};
  \node[fnote, anchor=west] at (2.67,-10.85) {only L.launch};
  \node[kept, anchor=west, align=center] at (0.1,-11.43) {keep $L$ owned, stop AI;\\radio recovery goes on};

  % (e) Repeated and old radio results reach the coordinator.
  \draw[pframe] (7.05,-8.35) rectangle ++(6.85,-3.55);
  \node[ptitle] at (7.1,-8.38) {(e) Repeated or old radio results};
  \foreach \x/\l in {9.1/Coordinator, 11.05/{cuPHY owner}, 12.9/{cuPHY recovery}} {
    \node[mhead] at (\x,-8.97) {\l};
    \draw[life] (\x,-9.15) -- (\x,-11.45);
  }
  \foreach \y/\src/\msg/\c/\res/\st in {
      -9.6/11.05/{TB1 result}/black/{applied}/gnote,
      -10.0/11.05/{TB1 result again}/dred/{no change}/nnote,
      -10.4/11.05/{old TB1 result}/dred/{no change}/nnote,
      -10.8/12.9/{TB3 decoded}/black/{commit TB3}/gnote,

    \node[\st, anchor=east] at (9.0,\y) {\res};
  }
